// ===============================================================================
// M1 - Ciudadanos, Organizaciones y Expedientes Digitales
// Colecciones de MongoDB: evento_outbox y evento_procesado
//
// Fuente: diagrama_db_desarrollo_de_apps_2.json (draw.io).
//
// Como ejecutarlo:
//     mongosh "mongodb://localhost:27017" --file mongodb_setup.txt
//
// o pegando el contenido en una sesion de mongosh.
//
// Por que Mongo y no Postgres: el payload de cada evento cambia por eventType
// (ver la tabla de eventos publicados en ARCHITECTURE.md), asi que no tiene un
// esquema fijo. Lo transaccional queda en Postgres.
//
// El schema validator de abajo valida el sobre del evento y deja libre el payload.
// ===============================================================================

// Base de datos del modulo. Coincide con MONGO_URI en application.yml.
db = db.getSiblingDB("ciudadanos_m1_eventos");


// ---------------------------------------------------------------------------
// evento_outbox
// Patron outbox: todo evento que M1 publica se escribe aca primero, dentro de
// la misma operacion de negocio, y despues lo toma el publisher. Si el publish
// falla, el evento sigue pendiente y se reintenta: no se pierde.
//
// aggregate_type + aggregate_id son el "subject" del contrato partido en dos
// (subject "citizens/42"  ->  aggregate_type "citizens", aggregate_id "42").
// El resto del contrato (occurredAt, producer, data, correlationId) va
// serializado dentro de payload.
// ---------------------------------------------------------------------------
db.createCollection("evento_outbox", {
    validator: {
        $jsonSchema: {
            bsonType: "object",
            required: ["_id", "aggregate_type", "aggregate_id", "event_type",
                       "payload", "created_at", "info", "details", "publicado"],
            properties: {
                // eventId del contrato: UUID de 36 caracteres, usado como _id
                // para que reinsertar el mismo evento sea un no-op.
                _id: {
                    bsonType: "string",
                    minLength: 36,
                    maxLength: 36,
                    description: "eventId (UUID v4)"
                },
                aggregate_type: {
                    bsonType: "string",
                    maxLength: 100,
                    description: "citizens, organizations, representations, ..."
                },
                aggregate_id: {
                    bsonType: "string",
                    maxLength: 100,
                    description: "id de la entidad afectada"
                },
                event_type: {
                    bsonType: "string",
                    maxLength: 100,
                    description: "citizenUpdated, organizationRegistered, ..."
                },
                payload:    { bsonType: "string", description: "evento completo serializado como JSON" },
                created_at: { bsonType: "date" },
                info:       { bsonType: "string", maxLength: 250 },
                details:    { bsonType: "string", maxLength: 2500 },

                // Agregados sobre el diagrama: sin esto la coleccion no funciona
                // como cola (no habria como saber que quedo pendiente).
                publicado:    { bsonType: "bool" },
                publicado_en: { bsonType: ["date", "null"] },

                _class: { bsonType: "string" }
            }
        }
    },
    validationLevel: "moderate",
    validationAction: "error"
});

// Cola de publicacion: pendientes, mas viejo primero.
db.evento_outbox.createIndex(
    { publicado: 1, created_at: 1 },
    { name: "ix_outbox_pendientes" }
);

// Consultas por tipo de evento y por entidad afectada.
db.evento_outbox.createIndex({ event_type: 1, created_at: -1 }, { name: "ix_outbox_tipo_fecha" });
db.evento_outbox.createIndex({ aggregate_type: 1, aggregate_id: 1 }, { name: "ix_outbox_aggregate" });


// ---------------------------------------------------------------------------
// evento_procesado
// Idempotencia del consumo. En el diagrama el eventId es la PK, asi que va como
// _id: reinsertar un evento ya procesado falla con E11000, y ESE es el chequeo.
// Un mismo evento no puede procesarse dos veces, ni siquiera por consumers
// distintos (si mas adelante hacen falta varios consumers sobre el mismo
// evento, la PK tiene que pasar a ser el par eventId + consumer_id).
// ---------------------------------------------------------------------------
db.createCollection("evento_procesado", {
    validator: {
        $jsonSchema: {
            bsonType: "object",
            required: ["_id", "consumer_id", "processed_at"],
            properties: {
                _id: {
                    bsonType: "string",
                    minLength: 36,
                    maxLength: 36,
                    description: "eventId del evento consumido"
                },
                consumer_id:  { bsonType: "string", maxLength: 100, description: "quien lo proceso" },
                processed_at: { bsonType: "date" },
                _class:       { bsonType: "string" }
            }
        }
    },
    validationLevel: "moderate",
    validationAction: "error"
});

db.evento_procesado.createIndex({ consumer_id: 1 }, { name: "ix_procesado_consumer" });

// Purga automatica a los 90 dias: pasado ese plazo ningun productor reenvia.
db.evento_procesado.createIndex(
    { processed_at: 1 },
    { name: "ttl_procesados", expireAfterSeconds: 7776000 }
);


// ---------------------------------------------------------------------------
// Verificacion
// ---------------------------------------------------------------------------
print("Colecciones: " + db.getCollectionNames().join(", "));
print("Indices evento_outbox:    " + db.evento_outbox.getIndexes().map(i => i.name).join(", "));
print("Indices evento_procesado: " + db.evento_procesado.getIndexes().map(i => i.name).join(", "));
